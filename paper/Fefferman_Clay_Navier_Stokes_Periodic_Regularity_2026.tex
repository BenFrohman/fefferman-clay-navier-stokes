\begin{abstract}
We present a self-contained Lean~4 formalization (\texttt{leanprover/lean4:v4.30.0-rc1}, Mathlib~4, 184 kernel-verified theorems across 5 modules with zero \texttt{sorry} and zero non-standard axioms outside \texttt{[propext, Classical.choice, Quot.sound]}) of Charles~L. Fefferman's official periodic 3D incompressible Navier--Stokes problem statement (\emph{Existence and Smoothness of the Navier--Stokes Equation}, Clay Mathematics Institute, 2006, Statement~(B) on $\mathbb{R}^3/\mathbb{Z}^3$) together with unconditional geometric/kinematic identities and explicit \textbf{conditional reductions} to Beale--Kato--Majda continuation.

\textbf{Scope and Honesty Statement:} This work does \textbf{not} solve Clay Problem~(B) or~(A). Specifically:
\begin{enumerate}
  \item We prove unconditionally that spatial $\mathbb{Z}^3$-periodicity excludes any non-zero \emph{global} linear background strain $u_{\mathrm{strain}}(x) = A x$ on $\mathbb{R}^3$ (\texttt{z3\_periodic\_linear\_eq\_zero}), including the global uniaxial Burgers background strain $A = \mathrm{diag}(-\alpha/2, -\alpha/2, \alpha)$ with $\alpha \neq 0$ (\texttt{burgersLinearStrain\_not\_z3\_periodic}). Crucially, torus periodicity rules out only a \emph{global} linear background strain: localized Burgers-like stretched vortex structures can still form locally on $\mathbb{T}^3 = \mathbb{R}^3/\mathbb{Z}^3$, so global linear strain exclusion does \textbf{not} yield unconditional Clay~(B).
  \item We prove unconditionally the pointwise Biot--Savart direction-alignment depletion identity (\texttt{biot\_savart\_det\_aligned\_eq\_zero}, \texttt{aligned\_vorticity\_stretching\_integrand\_zero}, \texttt{biot\_savart\_det\_difference\_identity}), the parabolic scale-invariance identities of 3D Navier--Stokes under $u_\lambda(x,t) = \lambda u(\lambda x, \lambda^2 t)$ (\texttt{navier\_stokes\_vorticity\_dt\_scaling\_exponent}, \texttt{navier\_stokes\_scaling\_homogeneity\_deg6}, \texttt{unnormalized\_quartic\_scaling\_mismatch}), and the cubic comparison ODE ceiling $M(t) \le \max(M(0), C_{\mathrm{CZ}}/\kappa'')$ (\texttt{comparison\_ode\_ceiling}) whenever a differentiable trajectory satisfies $M'(t) \le C_{\mathrm{CZ}} M(t)^2 - \kappa'' M(t)^3$.
  \item We formulate the analytic closure hypotheses \texttt{PeriodicDirectionDepletionBound~$\nu$}, \texttt{PeriodicTetherClosingBound~$\nu$}, and \texttt{PeriodicBKMContinuation~$\nu$} as explicit \texttt{Prop}-valued hypotheses (never as axioms) and prove the \textbf{conditional reductions} \texttt{clay\_B\_exact\_of\_periodic\_tether\_bound} and \texttt{clay\_B\_exact\_of\_torus\_direction\_depletion}. Whether arbitrary smooth periodic Navier--Stokes flows on $\mathbb{T}^3$ satisfy \texttt{PeriodicDirectionDepletionBound~$\nu$} or \texttt{PeriodicTetherClosingBound~$\nu$} remains an \textbf{open problem}.
\end{enumerate}
All Lean~4 proofs are audited in-kernel by \texttt{AxiomAudit.lean} and attested with an Ed25519-signed \texttt{titan-audit} certificate and \texttt{sealguard} Merkle-root provenance manifest.

\section{Fefferman's Official Periodic Navier--Stokes Formulation on $\mathbb{R}^3/\mathbb{Z}^3$}

Let $e_1, e_2, e_3$ denote the standard orthonormal basis of $\mathbb{R}^3$. Following Fefferman \cite{Fefferman2006}, a velocity field $u : \mathbb{R}^3 \to \mathbb{R}^3$ or time-dependent field $u : \mathbb{R}^3 \times [0,\infty) \to \mathbb{R}^3$ and pressure $p : \mathbb{R}^3 \times [0,\infty) \to \mathbb{R}$ are spatial $\mathbb{Z}^3$-periodic (\texttt{IsZ3Periodic} / \texttt{IsZ3PeriodicScalar}) if for all $x \in \mathbb{R}^3$, $t \ge 0$, and $1 \le j \le 3$,
\begin{equation}
  u(x + e_j, t) = u(x, t), \qquad p(x + e_j, t) = p(x, t).
  \label{eq:fefferman-periodic}
\end{equation}
The incompressible 3D Navier--Stokes system with viscosity $\nu > 0$ and zero external force ($f \equiv 0$) reads:
\begin{align}
  \frac{\partial u_i}{\partial t} + \sum_{j=1}^3 u_j \frac{\partial u_i}{\partial x_j} &= \nu \Delta u_i - \frac{\partial p}{\partial x_i} \qquad (x \in \mathbb{R}^3,\ t \ge 0,\ 1 \le i \le 3), \label{eq:fefferman-1}\\
  \operatorname{div} u = \sum_{i=1}^3 \frac{\partial u_i}{\partial x_i} &= 0 \qquad (x \in \mathbb{R}^3,\ t \ge 0), \label{eq:fefferman-2}\\
  u(x, 0) &= u^\circ(x) \qquad (x \in \mathbb{R}^3). \label{eq:fefferman-3}
\end{align}
In \texttt{FeffermanClayNS/FeffermanStatements.lean}, given smooth, divergence-free, $\mathbb{Z}^3$-periodic initial data $u^\circ \in C^\infty(\mathbb{R}^3/\mathbb{Z}^3)$ (\texttt{IsPeriodicSmoothInitialData}), a global smooth periodic solution (\texttt{IsPeriodicSmoothSolution}) satisfies \eqref{eq:fefferman-periodic}--\eqref{eq:fefferman-3} together with joint spacetime smoothness
\begin{equation}
  p, u_i \in C^\infty(\mathbb{R}^3 \times [0, \infty)) \qquad (\texttt{ContDiff } \mathbb{R}\ \top\ (\text{\texttt{uncurry }} u)).
  \label{eq:fefferman-11}
\end{equation}
Fefferman's Statement~(B) is encoded verbatim as \texttt{Clay\_B\_exact~$\nu$}, together with the energy-bounded variant \texttt{Clay\_B\_exact\_with\_energy~$\nu$} incorporating $\int_{\mathbb{T}^3} |u(x,t)|^2\,dx < C$ and the proved reduction \texttt{clay\_B\_exact\_of\_with\_energy}.

\subsection{Eliminating Step-Jump Vacuity in Local Existence Horizons}

A subtle pitfall in formalizing PDE existence horizons in Lean~4 is that a pointwise time derivative predicate $\partial_t u(t,x) = \cdots$ without continuity in $t$ permits discontinuous step-jump fields such as
\[
  u_{\mathrm{step}}(t, x) = \begin{cases} (0,0,0) & \text{if } t \le 0, \\ (1,0,0) & \text{if } t > 0, \end{cases}
\]
which satisfy $\partial_t u_{\mathrm{step}}(t,x) = 0 = \nu \Delta u_{\mathrm{step}} - \nabla p$ for every $t > 0$ while violating uniqueness from $u^\circ \equiv 0$. In \texttt{FeffermanStatements.lean}, we prove:
\begin{enumerate}
  \item \texttt{library\_huniq\_unsatisfiable\_at\_zero}: An unguarded local uniqueness hypothesis on \texttt{existenceTimes\_unpatched} is logically falsified (`False`) at $u^\circ \equiv 0$ by $u_{\mathrm{step}}$.
  \item \texttt{stepField\_not\_physical}: Upgrading the solution predicate to \texttt{IsPhysicalNSSolution} (requiring time continuity \texttt{ContinuousOn (fun t => u t x) (Set.Icc 0 T)} and spatial boundedness \texttt{BddAbove}) rules out $u_{\mathrm{step}}$.
  \item \texttt{repaired\_hInt}, \texttt{repaired\_huniq\_shift\_at\_zero}, and \texttt{repaired\_huniq\_at\_zero}: The repaired physical solution class is non-vacuously inhabited by the trivial flow $u \equiv 0$ and satisfies local uniqueness via the energy Gr\"onwall identity.
\end{enumerate}

\section{Global Linear Strain Exclusion on $\mathbb{T}^3$ and Its Exact Scope}

On all of $\mathbb{R}^3$, the classic Burgers vortex lives inside an unbounded uniaxial linear background strain
\begin{equation}
  u_{\mathrm{strain}}(x) = A x = \begin{pmatrix} -\alpha/2 & 0 & 0 \\ 0 & -\alpha/2 & 0 \\ 0 & 0 & \alpha \end{pmatrix} \begin{pmatrix} x_1 \\ x_2 \\ x_3 \end{pmatrix}, \qquad \operatorname{tr}(A) = -\frac{\alpha}{2} - \frac{\alpha}{2} + \alpha = 0.
  \label{eq:burgers-strain}
\end{equation}
In \texttt{FeffermanClayNS/TorusStrainExclusion.lean}, we prove that any $\mathbb{Z}^3$-periodic linear map on $\mathbb{R}^3$ vanishes identically:

\begin{theorem}[\texttt{z3\_periodic\_linear\_eq\_zero} and \texttt{burgersLinearStrain\_not\_z3\_periodic}]
Let $L : \mathbb{R}^3 \to_{\mathbb{R}} \mathbb{R}^3$ be any $\mathbb{R}$-linear map satisfying $\operatorname{IsZ3Periodic}(L)$, i.e., $L(x + e_j) = L(x)$ for all $x \in \mathbb{R}^3$ and $j \in \{1,2,3\}$. Then $L = 0$. In particular, for every $\alpha \neq 0$, the divergence-free global Burgers linear background strain \eqref{eq:burgers-strain} satisfies $\neg \operatorname{IsZ3Periodic}(\text{\texttt{burgersLinearStrain }}\alpha)$.
\end{theorem}
\begin{proof}
By linearity and periodicity at $x = 0$, $L(e_j) = L(0 + e_j) - L(0) = 0$ for each standard basis vector $e_j$. Since $\{e_1, e_2, e_3\}$ spans $\mathbb{R}^3$, $L(x) = \sum_{j=1}^3 x_j L(e_j) = 0$ for all $x \in \mathbb{R}^3$. For $A = \mathrm{diag}(-\alpha/2, -\alpha/2, \alpha)$, evaluation at $e_3$ yields $(A e_3)_3 = \alpha \neq 0$.
\end{proof}

\begin{remark}[Crucial Scope Warning: Why Global Linear Strain Exclusion Does Not Solve Clay~(B)]
Theorem~\texttt{burgersLinearStrain\_not\_z3\_periodic} rules out only a \emph{global} linear background strain $u(x) = Ax$ across $\mathbb{R}^3/\mathbb{Z}^3$. On the compact torus $\mathbb{T}^3$, a smooth periodic velocity field $u$ with zero mean ($\int_{\mathbb{T}^3} u\,dx = 0$) can still exhibit a large, locally linear strain matrix $\nabla u(x_0) = A$ inside a small neighborhood $B_r(x_0) \subset \mathbb{T}^3$, with the return flow closing smoothly elsewhere in the periodic box. Consequently, localized Burgers-like vortex tubes and sheets can still form locally on $\mathbb{T}^3$, and ruling out global linear strain does \textbf{not} yield an unconditional proof of Clay~(B).
\end{remark}

\section{Biot--Savart Direction Alignment Depletion and Parabolic Scaling Identities}

Taking the curl of \eqref{eq:fefferman-1} yields the 3D vorticity transport equation for $\omega = \nabla \times u$:
\begin{equation}
  \partial_t \omega + (u \cdot \nabla)\omega = (\omega \cdot \nabla)u + \nu \Delta \omega,
  \label{eq:vorticity-transport}
\end{equation}
and the pointwise enstrophy evolution along $\omega = |\omega|\hat{\xi}$ ($|\hat{\xi}| = 1$ on $\{\omega \neq 0\}$):
\begin{equation}
  \frac{1}{2}(\partial_t + u \cdot \nabla)|\omega|^2 = S(x)\omega(x) \cdot \omega(x) + \nu \Delta \frac{|\omega|^2}{2} - \nu |\nabla |\omega||^2 - \nu |\omega|^2 |\nabla \hat{\xi}|^2.
  \label{eq:enstrophy-direction}
\end{equation}
Under the Biot--Savart law, the vortex stretching quadratic form at $x$ has integrand proportional to the $3\times 3$ determinant
\[
  D(\omega(y), z, \omega(x)) = \det\bigl(\omega(y),\ z,\ \omega(x)\bigr) = \bigl(\omega(y) \times z\bigr) \cdot \omega(x), \qquad z = x - y.
\]
In \texttt{TorusStrainExclusion.lean}, we prove three unconditional algebraic identities for this kernel:
\begin{enumerate}
  \item \textbf{Collinear alignment vanishing (\texttt{biot\_savart\_det\_aligned\_eq\_zero}):} Whenever $\omega(y) = c\,\omega(x)$ for any scalar $c \in \mathbb{R}$, $\det(c\,\omega(x), z, \omega(x)) = 0$.
  \item \textbf{Complete stretching integrand vanishing under alignment (\texttt{aligned\_vorticity\_stretching\_integrand\_zero}):} The full singular stretching integrand $3\frac{(z \cdot \omega(x))\,\det(\omega(y), z, \omega(x))}{|z|^5}$ vanishes identically whenever $\omega(y) = c\,\omega(x)$.
  \item \textbf{Exact direction-difference decomposition (\texttt{biot\_savart\_det\_difference\_identity}):} For all $\omega_x, \omega_y, z \in \mathbb{R}^3$,
  \begin{equation}
    \det(\omega_y, z, \omega_x) = \det(\omega_y - \omega_x, z, \omega_x),
    \label{eq:det-diff}
  \end{equation}
  demonstrating that vortex stretching at $x$ is driven exclusively by deviations of $\omega(y)$ from the direction of $\omega(x)$ (Constantin--Fefferman geometric depletion \cite{ConstantinFefferman1993}).
\end{enumerate}

Furthermore, under parabolic Navier--Stokes rescaling $u_\lambda(x,t) = \lambda u(\lambda x, \lambda^2 t)$, the vorticity scales as $\omega_\lambda(x,t) = \lambda^2 \omega(\lambda x, \lambda^2 t)$ and the direction gradient scales as $(\nabla \hat{\xi})_\lambda(x,t) = \lambda (\nabla \hat{\xi})(\lambda x, \lambda^2 t)$. We prove in \texttt{navier\_stokes\_scaling\_homogeneity\_deg6} and \texttt{unnormalized\_quartic\_scaling\_mismatch} that the viscous direction-dissipation density $\nu |\omega|^2 |\nabla \hat{\xi}|^2$ scales homogeneously of degree $6$ in $\lambda$:
\begin{equation}
  \nu (\lambda^2 |\omega|)^2 (\lambda |\nabla \hat{\xi}|)^2 = \lambda^6 \nu |\omega|^2 |\nabla \hat{\xi}|^2,
  \label{eq:lambda6}
\end{equation}
matching $\partial_t |\omega_\lambda|^2 \sim \lambda^6$, whereas an unnormalized quartic damping $\kappa |\omega|^4$ scales as $\lambda^8 \neq \lambda^6$ for $\lambda \notin \{0,1\}$.

\section{Cubic Comparison ODE Ceiling and Conditional Reductions to \texttt{Clay\_B\_exact}}

In \texttt{FeffermanClayNS/HighSetDepletionAndBKM.lean}, we first prove the algebraic and differential bridges showing that if at the spatial maximum $M(t) = \|\omega(t)\|_\infty = |\omega(t, x_*(t))|$ the high-vorticity direction gradient satisfies the scale-invariant coercivity lower bound $|\nabla \hat{\xi}(x_*)|^2 \ge c_{\mathrm{dir}} M(t)$ with $c_{\mathrm{dir}} > 0$, then the viscous direction-dissipation term $-\nu M(t)^2 |\nabla \hat{\xi}(x_*)|^2 \le -\nu c_{\mathrm{dir}} M(t)^3$ produces the cubic differential inequality (\texttt{scale\_invariant\_cubic\_rate\_of\_direction\_coercivity}):
\begin{equation}
  M'(t) \le C_{\mathrm{CZ}} M(t)^2 - \kappa'' M(t)^3, \qquad \kappa'' = \nu c_{\mathrm{dir}} > 0.
  \label{eq:cubic-ode}
\end{equation}

We then prove unconditionally the global ceiling theorem for any differentiable function $M : \mathbb{R} \to \mathbb{R}$ obeying \eqref{eq:cubic-ode}:

\begin{theorem}[\texttt{comparison\_ode\_ceiling}]
Let $M : \mathbb{R} \to \mathbb{R}$ be differentiable at every $t \in [0,T]$ and satisfy $M'(t) \le C_{\mathrm{CZ}} M(t)^2 - \kappa'' M(t)^3$ on $[0,T]$ with $C_{\mathrm{CZ}} > 0$ and $\kappa'' > 0$. Then for every $t \in [0,T]$,
\begin{equation}
  M(t) \le \max\!\left(M(0),\, \frac{C_{\mathrm{CZ}}}{\kappa''}\right).
  \label{eq:ceiling-bound}
\end{equation}
\end{theorem}
\begin{proof}
Let $Y = \max(M(0), C_{\mathrm{CZ}}/\kappa'') > 0$. For any $\varepsilon > 0$, define the constant barrier $B(s) = Y + \varepsilon$. At any contact point $s \in [0,t]$ where $M(s) = Y + \varepsilon$, we have $M(s) > C_{\mathrm{CZ}}/\kappa'' > 0$, so $\kappa'' M(s) > C_{\mathrm{CZ}}$ and thus
\[
  M'(s) \le M(s)^2\bigl(C_{\mathrm{CZ}} - \kappa'' M(s)\bigr) < 0 = B'(s).
\]
Applying Mathlib's first-exit-time comparison lemma \texttt{image\_le\_of\_deriv\_right\_lt\_deriv\_boundary} yields $M(t) \le Y + \varepsilon$ for every $\varepsilon > 0$, and \texttt{le\_of\_forall\_pos\_lt\_add} concludes $M(t) \le Y$.
\end{proof}

Finally, we state the two open analytic hypotheses and the conditional reductions:
\begin{hypothesis}[\texttt{PeriodicDirectionDepletionBound~$\nu$} --- OPEN]
For every smooth, divergence-free, $\mathbb{Z}^3$-periodic initial datum $u^\circ$, there exist $C_{\mathrm{CZ}}, \kappa'' > 0$ such that along any local smooth periodic solution on $[0,T]$, $M(t) = \|\omega(t)\|_\infty$ is differentiable and obeys $M'(t) \le C_{\mathrm{CZ}} M(t)^2 - \kappa'' M(t)^3$ on $[0,T]$.
\end{hypothesis}

\begin{hypothesis}[\texttt{PeriodicTetherClosingBound~$\nu$} --- OPEN]
For every smooth, divergence-free, $\mathbb{Z}^3$-periodic initial datum $u^\circ$, there exists $Y \ge 0$ such that $\|\omega(t)\|_\infty \le Y$ for all $t \in [0,T]$ along any local smooth periodic solution on $[0,T]$.
\end{hypothesis}

\begin{theorem}[Conditional Reductions: \texttt{periodicTetherClosingBound\_of\_directionDepletion}, \texttt{clay\_B\_exact\_of\_periodic\_tether\_bound}, and \texttt{clay\_B\_exact\_of\_torus\_direction\_depletion}]
\leavevmode
\begin{enumerate}
  \item \texttt{PeriodicDirectionDepletionBound~$\nu$} implies \texttt{PeriodicTetherClosingBound~$\nu$} with explicit ceiling $Y = \max(\|\omega_0\|_\infty, C_{\mathrm{CZ}}/\kappa'')$.
  \item Conditional on \texttt{PeriodicTetherClosingBound~$\nu$} (OPEN) and \texttt{PeriodicBKMContinuation~$\nu$} (classical Kato + Beale--Kato--Majda continuation \cite{Kato1972,BKM1984}), \texttt{Clay\_B\_exact~$\nu$} holds.
  \item Conditional on \texttt{PeriodicDirectionDepletionBound~$\nu$} (OPEN) and \texttt{PeriodicBKMContinuation~$\nu$}, \texttt{Clay\_B\_exact~$\nu$} holds.
\end{enumerate}
\end{theorem}
