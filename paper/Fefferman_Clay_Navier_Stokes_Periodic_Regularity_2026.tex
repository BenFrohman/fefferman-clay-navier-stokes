\begin{abstract}
We present a self-contained Lean~4 formalization (\texttt{leanprover/lean4:v4.30.0-rc1}, Mathlib~4, 236 theorem/lemma declarations in source across 7 modules with zero \texttt{sorry} and zero non-standard axioms outside \texttt{[propext, Classical.choice, Quot.sound]}) of Charles~L. Fefferman's official periodic 3D incompressible Navier--Stokes problem statement (\emph{Existence and Smoothness of the Navier--Stokes Equation}, Clay Mathematics Institute, 2006, Statement~(B) on $\mathbb{R}^3/\mathbb{Z}^3$) together with unconditional geometric/kinematic identities, spatial-maximum Bochner--polar reduction, and explicit \textbf{conditional reductions} to Beale--Kato--Majda continuation. In \texttt{FeffermanClayNS/FinalProductTheorem.lean}, five proved components (Q$_1$--Q$_3$ unconditional; Q$_4$ and Q$_5$ implications from stated hypotheses) are packaged into the single 5-conjunct master product theorem \texttt{fefferman\_periodic\_ns\_exact\_reduction\_product\_theorem} ($Q_1(\nu) \wedge Q_2 \wedge Q_3 \wedge Q_4 \wedge Q_5(\nu)$).
\end{abstract}

{\sloppy\renewcommand{\UrlBreaks}{\do\_\do\.}\paragraph{Erratum (v1.5.0, November 11, 2026).} In v1.0--v1.4 the open hypothesis of Q$_5$, namely \path{PeriodicDirectionDepletionBound} $\nu$, required a two-sided derivative (\path{DifferentiableAt}) of $M(t)=\|\omega(t)\|_\infty$ at every $t\in[0,T]$, including $t=0$. But the solution predicate \path{IsPeriodicSmoothSolutionOn} does not constrain $u(t)$ for $t<0$, so the path \path{jumpPath}, equal to the zero solution on $[0,\infty)$ and to a linear shear with nonzero constant vorticity for $t<0$, is a solution on $[0,1]$ whose $M$ jumps at $0$. The kernel-checked theorem \path{old_periodicDirectionDepletionBound_unsatisfiable} proves $\neg$\,\path{PeriodicDirectionDepletionBoundOriginal}~$\nu$ for every $\nu$ (the old form is kept under that name). Hence Q$_5$ and \path{clay_B_exact_of_torus_direction_depletion} were vacuously true in v1.0--v1.4. From v1.5.0 the hypothesis requires continuity of $M$ on $[0,T]$ and a right derivative $d$ at each $s\in[0,T)$ with $d\le C_{\mathrm{CZ}}M(s)^2-\kappa''M(s)^3$; the ODE ceiling is \path{comparison_ode_ceiling_right}. The regression \path{jumpPath_oneSided_depletion_clause} shows the counterexample path satisfies the corrected trajectory clause; it does not show the corrected hypothesis holds in general, which remains open. All statements below refer to the corrected definition. Nothing here resolves the Clay problem.\par}

\section{Introduction and Single-Sentence Final Product Form ($Q_1(\nu) \wedge Q_2 \wedge Q_3 \wedge Q_4 \wedge Q_5(\nu)$)}

In his official Clay Mathematics Institute problem description \cite{Fefferman2006}, Charles~L. Fefferman formulates the periodic existence and smoothness problem (Statement~(B)) for the three-dimensional incompressible Navier--Stokes equations on $\mathbb{T}^3 = \mathbb{R}^3/\mathbb{Z}^3$. For the unmodified Navier--Stokes PDE ($\kappa = 0$ in the dynamics), Lean's kernel cannot derive $\langle\omega,(\omega\cdot\nabla)u\rangle + \nu\langle\omega,\Delta\omega\rangle \le C_{\mathrm{CZ}}M^3 - \kappa'' M^4$ at a spatial maximum without either taking the high-set direction-curvature bound $|\nabla\hat{\xi}(x_{\max})|^2 \ge c_{\mathrm{dir}} M$ as an explicit hypothesis or proving that open PDE estimate from scratch. Accordingly, this repository states \textbf{only the theorems it proves explicitly and unconditionally}, packaged in \texttt{FeffermanClayNS/FinalProductTheorem.lean} into a single 5-conjunct product theorem.

\subsection{Single-Sentence English Prose Statement}
\begin{theorem}[Master Product Theorem: \texttt{fefferman\_periodic\_ns\_exact\_reduction\_product\_theorem}]
For every viscosity $\nu \in \mathbb{R}$ on the 3-torus $\mathbb{T}^3 = \mathbb{R}^3/\mathbb{Z}^3$, \textbf{(Q$_1$)} spatial $\mathbb{Z}^3$-periodicity is equivalent to invariance under the three unit coordinate shifts $f(x+e_j)=f(x)$, and Fefferman's energy-bounded periodic statement implies his exact periodic statement ($\mathrm{Clay\_B\_exact\_with\_energy}(\nu) \implies \mathrm{Clay\_B\_exact}(\nu)$); \textbf{(Q$_2$)} every $\mathbb{Z}^3$-periodic linear map $L : \mathbb{R}^3 \to \mathbb{R}^3$ vanishes identically ($L = 0$), excluding any non-zero global linear Burgers background strain $\mathrm{diag}(-\alpha/2,-\alpha/2,\alpha)x$ ($\alpha \neq 0$) on $\mathbb{T}^3$; \textbf{(Q$_3$)} the Biot--Savart vortex-stretching determinant obeys $\det(\omega(y),z,\omega(x)) = \det(\omega(y)-\omega(x),z,\omega(x))$ and vanishes under collinear alignment $\omega(y)=c\,\omega(x)$, while under parabolic rescaling $u_\lambda(x,t)=\lambda u(\lambda x,\lambda^2 t)$ both cubic stretching $C_{\mathrm{CZ}}|\omega_\lambda|^3$ and viscous direction-curvature dissipation $\nu|\omega_\lambda|^2|\nabla\hat{\xi}_\lambda|^2$ scale homogeneously as $\lambda^6$; \textbf{(Q$_4$)} at a spatial maximum $x_{\max}$ where $\langle\omega,(\omega\cdot\nabla)u\rangle \le C_{\mathrm{CZ}}M^3$, $\nu\langle\omega,\Delta\omega\rangle \le -\nu M^2|\nabla\hat{\xi}|^2$, and high-set direction-curvature coercivity $|\nabla\hat{\xi}(x_{\max})|^2 \ge c_{\mathrm{dir}}M$ holds, the maximum vorticity rate satisfies $M'(t) \le C_{\mathrm{CZ}}M(t)^2 - (\nu c_{\mathrm{dir}})M(t)^3$, which forces the uniform Lyapunov ceiling $M(t) \le \max(M(0), C_{\mathrm{CZ}}/(\nu c_{\mathrm{dir}}))$; and \textbf{(Q$_5$)} whenever a periodic flow class satisfies the high-set direction-curvature depletion hypothesis $\mathrm{PeriodicDirectionDepletionBound}(\nu)$ (one-sided form, v1.5.0) and classical periodic Beale--Kato--Majda continuation $\mathrm{PeriodicBKMContinuation}(\nu)$, Fefferman's periodic statement $\mathrm{Clay\_B\_exact}(\nu)$ holds.
\end{theorem}

\subsection{Exact Mathematical Product Form ($Q_1(\nu) \wedge Q_2 \wedge Q_3 \wedge Q_4 \wedge Q_5(\nu)$)}
\begin{equation}
\label{eq:master-product-clayb}
\begin{aligned}
\mathcal{Q}_{\mathrm{ClayB}}(\nu) \;\;:\Longleftrightarrow\;\;
&\underbrace{\Bigl(\bigl(\forall f:\mathbb{R}^3\to\mathbb{R}^3,\; \mathrm{IsZ3Periodic}(f) \iff \forall x,\, j\in\{1,2,3\},\; f(x+e_j)=f(x)\bigr) \;\wedge\; \bigl(\mathrm{Clay\_B\_exact\_with\_energy}(\nu) \implies \mathrm{Clay\_B\_exact}(\nu)\bigr)\Bigr)}_{Q_1(\nu):\;\text{Exact Fefferman Periodic Unit-Shift Equivalence \& Energy Implication}} \\
&\wedge\; \underbrace{\Bigl(\bigl(\forall L:\mathbb{R}^3\to_{\mathbb{R}}\mathbb{R}^3,\; \mathrm{IsZ3Periodic}(L) \implies L=0\bigr) \;\wedge\; \bigl(\forall \alpha\neq 0,\; \neg\mathrm{IsZ3Periodic}(\mathrm{diag}(-\tfrac{\alpha}{2},-\tfrac{\alpha}{2},\alpha))\bigr)\Bigr)}_{Q_2:\;\text{Torus Exclusion of Global Linear Burgers Strain}} \\
&\wedge\; \underbrace{\Bigl(\det(\omega_y,z,\omega_x)=\det(\omega_y-\omega_x,z,\omega_x) \;\wedge\; \det(c\omega_x,z,\omega_x)=0 \;\wedge\; C_{\mathrm{CZ}}(\lambda^2|\omega|)^3 - \nu(\lambda^2|\omega|)^2(\lambda|\nabla\hat{\xi}|)^2 = \lambda^6\bigl(C_{\mathrm{CZ}}|\omega|^3 - \nu|\omega|^2|\nabla\hat{\xi}|^2\bigr)\Bigr)}_{Q_3:\;\text{Biot--Savart Alignment Depletion \& }\lambda^6\text{ Parabolic Scaling Homogeneity}} \\
&\wedge\; \underbrace{\Bigl(\bigl(\mathrm{Stretch}\le C_{\mathrm{CZ}}M^3 \wedge \mathrm{Diff}\le -\nu M^2|\nabla\hat{\xi}|^2 \wedge |\nabla\hat{\xi}|^2\ge c_{\mathrm{dir}}M \implies M'\le C_{\mathrm{CZ}}M^2 - \nu c_{\mathrm{dir}}M^3\bigr) \;\wedge\; \bigl(M'\le C_{\mathrm{CZ}}M^2 - \kappa'' M^3 \implies M(t)\le\max(M(0),\tfrac{C_{\mathrm{CZ}}}{\kappa''})\bigr)\Bigr)}_{Q_4:\;\text{Spatial-Maximum Bochner--Polar Reduction \& Cubic ODE Ceiling}} \\
&\wedge\; \underbrace{\Bigl(\bigl(\mathrm{PeriodicDirectionDepletionBound}(\nu) \;\wedge\; \mathrm{PeriodicBKMContinuation}(\nu)\bigr) \implies \mathrm{Clay\_B\_exact}(\nu)\Bigr)}_{Q_5(\nu):\;\text{Exact Conditional Implication to Fefferman's }\mathrm{Clay\_B\_exact}(\nu)}.
\end{aligned}
\end{equation}
